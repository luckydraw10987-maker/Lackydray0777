<script>
let players = [
"Kkk", "KKKkkkkkkkkkh", "Player #3", "Player #4", "Player #5",
"Player #6", "Player #7", "Player #8", "Player #9", "Player #10",
"Player #11", "Player #12", "Player #13", "Player #14", "Player #15",
"Player #16", "Player #17", "Player #18", "Player #19", "Player #20"
 ];

let currentReplaceIndex = 0;

const playerSlotsContainer = document.getElementById('playerSlotsContainer');
const forcedWinnerSelect = document.getElementById('forcedWinnerSelect');
const playBtn = document.getElementById('playBtn');
const nameInput = document.getElementById('nameInput');
const addBtn = document.getElementById('addBtn');
const box1 = document.getElementById('box1');
const box2 = document.getElementById('box2');
const box3 = document.getElementById('box3');
const winnerNameDisplay = document.getElementById('winnerNameDisplay');
const winnerDisplayBox = document.getElementById('winnerDisplayBox');
const winnerLabelText = document.getElementById('winnerLabelText');

function renderInterface() {
playerSlotsContainer.innerHTML = '';
const currentSelected = forcedWinnerSelect.value;
forcedWinnerSelect.innerHTML = '<option value="random">Random</option>';

players.forEach((player, index) => {
const slot = document.createElement('div');
slot.className = 'player-slot';
slot.id = player-slot-${index}; slot.innerText ={player}`;
playerSlotsContainer.appendChild(slot);

const option = document.createElement('option');
option.value = index;
option.innerText = Pos ${index + 1}: ${player};
forcedWinnerSelect.appendChild(option);
});

forcedWinnerSelect.value = currentSelected;
}

function startCelebration() {
const colors = ['#ffd700', '#2ea043', '#f85149', '#58a6ff', '#ff9800', '#e14eca'];
for (let i = 0; i < 120; i++) {
setTimeout(() => {
const confetti = document.createElement('div');
confetti.className = 'confetti';
confetti.style.left = Math.random() * 100 + 'vw';
confetti.style.backgroundColor = colors[Math.floor(Math.random() * colors.length)];
confetti.style.width = Math.random() * 8 + 6 + 'px';
confetti.style.height = Math.random() * 12 + 6 + 'px';
confetti.style.animationDuration = Math.random() * 2 + 1.5 + 's';
confetti.style.opacity = Math.random();

document.body.appendChild(confetti);

setTimeout(() => { confetti.remove(); }, 3500);
}, i * 25);
}
}

addBtn.addEventListener('click', () => {
const newName = nameInput.value.trim();
if (newName !== "") {
players[currentReplaceIndex] = newName;
currentReplaceIndex++;
if (currentReplaceIndex >= 20) {
currentReplaceIndex = 0;
}
renderInterface();
nameInput.value = "";
nameInput.focus();
}
});

nameInput.addEventListener('keypress', (e) => {
if (e.key === 'Enter') {
addBtn.click();
}
});

playBtn.addEventListener('click', () => {
document.querySelectorAll('.player-slot').forEach(el => el.classList.remove('winner-highlight'));
winnerDisplayBox.classList.remove('winner-celebrate');

playBtn.disabled = true;

const selectedValue = forcedWinnerSelect.value;
let winnerIndex;

if (selectedValue === 'random') {
winnerIndex = Math.floor(Math.random() * players.length);
} else {
winnerIndex = parseInt(selectedValue);
}

let counter = 0;
const totalTicks = 400; // ১০০ মিলি-সেকেন্ড পর পর মোট ৪০০ বার ঘুরবে (৪০ সেকেন্ড)

const interval = setInterval(() => {
box1.innerText = Math.floor(Math.random() * 10);
box2.innerText = Math.floor(Math.random() * 10);
box3.innerText = Math.floor(Math.random() * 10);
counter++;

// লাইভ কাউন্টডাউন টেক্সট আপডেট
let secondsLeft = ((totalTicks - counter) * 0.1).toFixed(1);
winnerLabelText.innerText = "⏳ SPINNING...";
winnerNameDisplay.innerText = CHOOSING IN ${secondsLeft}s`;

if (counter >= totalTicks) {
clearInterval(interval);

const displayNum = String(winnerIndex + 1).padStart(3, '0');
box1.innerText = displayNum[0];
box2.innerText = displayNum[1];
box3.innerText = displayNum[2];

winnerLabelText.innerText = "🎉 WINNER CONGRATS 🎉";
winnerNameDisplay.innerText = players[winnerIndex].toUpperCase();

winnerDisplayBox.classList.add('winner-celebrate');
startCelebration();

document.getElementById(player-slot-${winnerIndex}).classList.add('winner-highlight'); document.getElementById(player-slot-${winnerIndex}`).scrollIntoView({ behavior: 'smooth', block: 'nearest' });

playBtn.disabled = false;
}
}, 100);
});

renderInterface();
</script>
